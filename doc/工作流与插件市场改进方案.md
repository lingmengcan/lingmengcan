#### 1.1 断点调试系统

**技术实现方案**：

```typescript
// 后端：断点管理服务
// service/src/modules/workflow/services/breakpoint.service.ts
@Injectable()
export class BreakpointService {
  private breakpoints: Map<string, Set<string>> = new Map(); // workflowId -> nodeIds

  /**
   * 设置断点
   */
  setBreakpoint(workflowId: string, nodeId: string): void {
    if (!this.breakpoints.has(workflowId)) {
      this.breakpoints.set(workflowId, new Set());
    }
    this.breakpoints.get(workflowId)!.add(nodeId);
  }

  /**
   * 移除断点
   */
  removeBreakpoint(workflowId: string, nodeId: string): void {
    this.breakpoints.get(workflowId)?.delete(nodeId);
  }

  /**
   * 检查是否在断点处
   */
  isBreakpoint(workflowId: string, nodeId: string): boolean {
    return this.breakpoints.get(workflowId)?.has(nodeId) || false;
  }
}

// 执行引擎集成断点
// service/src/modules/workflow/engine/workflow-execution-engine.ts
async executeNode(node: WorkflowNode, context: WorkflowContext, ...) {
  // 检查断点
  if (this.breakpointService.isBreakpoint(context.workflowId, node.id)) {
    // 暂停执行，等待调试命令
    await this.pauseForDebug(node, context);
  }
  
  // 执行节点
  const result = await this.nodeExecutor.execute(node, context);
  return result;
}
```

interface PerformanceMetrics {

// 性能分析服务

    const v1 = await this.getVersion(workflowId, version1);
    const v2 = await this.getVersion(workflowId, version2);
    
    return {
      added: this.diffNodes(v1.config.nodes, v2.config.nodes, 'added'),
      removed: this.diffNodes(v1.config.nodes, v2.config.nodes, 'removed'),
      modified: this.diffNodes(v1.config.nodes, v2.config.nodes, 'modified')
    };
  }

@WebSocketGateway({ namespace: '/workflow' })
export class WorkflowCollaborationGateway {
  @WebSocketServer()
  server: Server;

  @SubscribeMessage('join-workflow')
  handleJoinWorkflow(client: Socket, payload: { workflowId: string }) {
    client.join(`workflow:${payload.workflowId}`);
    // 通知其他用户
    client.to(`workflow:${payload.workflowId}`).emit('user-joined', {
      userId: client.data.userId,
      userName: client.data.userName
    });
  }

  @SubscribeMessage('node-update')
  handleNodeUpdate(client: Socket, payload: { workflowId: string, node: any }) {
    // 广播节点更新（排除发送者）
    client.to(`workflow:${payload.workflowId}`).emit('node-updated', payload);
  }

  @SubscribeMessage('cursor-move')
  handleCursorMove(client: Socket, payload: { workflowId: string, position: any }) {
    client.to(`workflow:${payload.workflowId}`).emit('cursor-moved', {
      userId: client.data.userId,
      position: payload.position
    });
  }
}

// service/src/modules/plugin/services/marketplace.service.ts
@Injectable()
export class MarketplaceService {
  async publishToMarketplace(pluginId: string, userId: string) {
    const plugin = await this.pluginService.findOne(pluginId);
    
    // 验证插件
    await this.validatePlugin(plugin);
    
    // 发布到市场
    plugin.marketplaceStatus = 1;
    await this.pluginService.update(plugin, userId);
    
    return plugin;
  }

  async getMarketplacePlugins(query: MarketplaceQueryDto) {
    const queryBuilder = this.pluginRepository
      .createQueryBuilder('plugin')
      .where('plugin.marketplaceStatus = :status', { status: 1 })
      .andWhere('plugin.status = :active', { active: 0 });
    
    // 搜索
    if (query.keyword) {
      queryBuilder.andWhere(
        '(plugin.pluginName LIKE :keyword OR plugin.description LIKE :keyword)',
        { keyword: `%${query.keyword}%` }
      );
    }
    
    // 分类筛选
    if (query.category) {
      queryBuilder.andWhere('plugin.pluginType = :category', { category: query.category });
    }
    
    // 排序
    if (query.sortBy === 'rating') {
      queryBuilder.orderBy('plugin.rating', 'DESC');
    } else if (query.sortBy === 'downloads') {
      queryBuilder.orderBy('plugin.downloadCount', 'DESC');
    }
    
    return await queryBuilder.getManyAndCount();
  }

  async installPlugin(pluginId: string, userId: string) {
    const plugin = await this.findOne(pluginId);
    
    // 检查依赖
    await this.checkDependencies(plugin);
    
    // 安装插件
    const installedPlugin = await this.createInstalledPlugin(plugin, userId);
    
    // 记录下载
    await this.recordDownload(pluginId, userId);
    
    return installedPlugin;
  }
}

// service/src/modules/plugin/services/sandbox.service.ts
@Injectable()
export class SandboxService {
  private vm: NodeVM;

  constructor() {
    this.vm = new NodeVM({
      sandbox: {
        // 允许的全局对象
        console: console,
        Buffer: Buffer,
        // 禁止的
        require: undefined,
        process: undefined,
        fs: undefined
      },
      require: {
        external: ['axios'], // 允许的外部模块
        builtin: [] // 禁止内置模块
      },
      timeout: 5000, // 执行超时
      memoryLimit: 128 * 1024 * 1024 // 内存限制 128MB
    });
  }

  async executePlugin(pluginCode: string, context: PluginContext): Promise<any> {
    try {
      const result = await this.vm.run(pluginCode, {
        context: this.sanitizeContext(context)
      });
      return result;
    } catch (error) {
      throw new PluginExecutionError(error.message);
    }
  }

  private sanitizeContext(context: PluginContext): any {
    // 清理上下文，移除敏感信息
    return {
      inputs: context.inputs,
      variables: context.variables,
      // 提供安全的 API
      http: this.createSafeHttpClient(context),
      logger: context.logger
    };
  }
}
